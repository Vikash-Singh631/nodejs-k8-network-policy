apiVersion: networking:k8s.io/v1
kind: NetworkPolicy
metadata:
  name: my-app-network-policy
spec:
  podSelector:
    matchLabels:
      app: my-app
  policyTypes:
    - Ingress
    - Egress
  ingress:
    - from:
        - podSelector: {}        # Allow traffic from any pod in the same namespace
        - podSelector:
	    matchLabels:
              app: trusted       # Specific trusted pod
  egress:
    - to:
        -podSelector: {}         # Allows traffic to any pod in the same namespace
